		end;	// an AI player, not a person

// First steps on a map: save it.
function	script	F_ChronWKDiscover	{
	.@m$ = getarg(0);
	if (callfunc("F_ChronWKKind", .@m$) == "")
		return;
	if (inarray(NPCP_WM$, .@m$) >= 0)
		return;
	.@n = getarraysize(NPCP_WM$);
	NPCP_WM$[.@n] = .@m$;
	NPCP_WX[.@n] = getarg(1);
	NPCP_WY[.@n] = getarg(2);
	dispbottom "[Waykeeper] New place: " + callfunc("F_ChronWKPretty", .@m$) + ". Any Waykeeper can now send you here.", 0x99DDFF;
	return;
}

// town / field / dungeon, or "" for maps that are no place to warp to.
function	script	F_ChronWKKind	{
	.@m$ = getarg(0);
	if (compare(.@m$, "@") || compare(.@m$, "#"))
		return "";
	// A map that's gone (a temporary copy like "0674#000001", or a removed
	// mod's map): getmapflag fails on it and kills the whole menu.
	if (getmapusers(.@m$) < 0)
		return "";
	if (getmapflag(.@m$, MF_PVP) || getmapflag(.@m$, MF_GVG) || getmapflag(.@m$, MF_BATTLEGROUND))
		return "";
	// The Juperos elevator rooms: empty dead ends outside the lever event.
	if (.@m$ == "jupe_ele" || .@m$ == "jupe_ele_r")
		return "";
	// Towns by name first: Morroc lacks the town flag in pre-renewal.
	setarray .@towns$, "prontera", "izlude", "geffen", "payon", "morocc", "alberta", "aldebaran", "yuno", "comodo", "umbala", "amatsu", "gonryun", "louyang", "ayothaya", "jawaii", "xmas", "niflheim", "einbroch", "einbech", "lighthalzen", "hugel", "rachel", "veins", "moscovia", "brasilis", "manuk", "splendide";
	if (inarray(.@towns$, .@m$) >= 0)
		return "town";
	if (getmapflag(.@m$, MF_TOWN))
		return "town";
	if (.@m$ == "pay_arche" || .@m$ == "izlu2dun" || .@m$ == "moc_ruins" || .@m$ == "thor_camp" || .@m$ == "alb2trea" || .@m$ == "mid_camp")
		return "town";
	if (compare(.@m$, "fild") || compare(.@m$, "mjolnir_") || compare(.@m$, "_gld") || .@m$ == "glast_01")
		return "field";
	// Glast Heim before the filters below: "_cas" and "church" are meant for
	// guild castles and town interiors, not gl_cas01 or gl_church.
	if (compare(.@m$, "sphinx") || compare(.@m$, "gl_"))
		return "dungeon";
	if (compare(.@m$, "_in") || compare(.@m$, "in_") || compare(.@m$, "job") || compare(.@m$, "que_") || compare(.@m$, "_cas") || compare(.@m$, "arena") || compare(.@m$, "jail") || compare(.@m$, "sec_") || compare(.@m$, "castle") || compare(.@m$, "church") || compare(.@m$, "_room") || compare(.@m$, "new_") || compare(.@m$, "ordeal") || compare(.@m$, "_tt") || compare(.@m$, "turbo") || compare(.@m$, "quiz") || compare(.@m$, "airplane") || compare(.@m$, "airport") || compare(.@m$, "ship"))
		return "";
	return "dungeon";
}

// A readable map name: "Payon Cave 2", "Prontera".
function	script	F_ChronWKPretty	{
	.@m$ = getarg(0);
	.@i = inarray($@chrwk_tmap$, .@m$);
	if (.@i >= 0)
		return $@chrwk_tname$[.@i];
	.@num$ = "";
	.@len = getstrlen(.@m$);
	while (.@len > 0 && charat(.@m$, .@len - 1) >= "0" && charat(.@m$, .@len - 1) <= "9") {
		.@num$ = charat(.@m$, .@len - 1) + .@num$;
		.@len--;
	}
	for (.@i = 0; .@i < getarraysize($@chrwk_pfx$); .@i++) {
		if (compare(.@m$, $@chrwk_pfx$[.@i])) {
			if (.@num$ != "")
				return $@chrwk_pname$[.@i] + " " + atoi(.@num$);
			return $@chrwk_pname$[.@i];
		}
	}
	return .@m$;
}

-	script	ChronWaykeeper	-1,{
	.@n$ = "[Waykeeper]";
	mes .@n$;
	mes "I can send you anywhere you have already set foot. The road remembers you.";
	next;
	.@k = select("Towns:Fields:Dungeons:Cancel");
	if (.@k == 4)
		close;
	.@want$ = "dungeon";
	if (.@k == 1)
		.@want$ = "town";
	else if (.@k == 2)
		.@want$ = "field";
	.@c = 0;
	.@opt$ = "";
	// Each place is named through tables of ~100 entries: a few dozen places
	// pass rAthena's runaway-loop guard ("infinity loop!") without freeloop.
	freeloop(1);
	for (.@i = 0; .@i < getarraysize(NPCP_WM$); .@i++) {
		if (callfunc("F_ChronWKKind", NPCP_WM$[.@i]) != .@want$)
			continue;
		.@m$[.@c] = NPCP_WM$[.@i];
		.@x[.@c] = NPCP_WX[.@i];
		.@y[.@c] = NPCP_WY[.@i];
		.@opt$ += callfunc("F_ChronWKPretty", NPCP_WM$[.@i]) + ":";
		.@c++;
	}
	freeloop(0);
	if (!.@c) {
		mes .@n$;
		mes "You have not set foot in any yet. Go and see the world; I will remember every place you reach.";
		close;
	}
	.@s = select(.@opt$ + "Cancel") - 1;
	if (.@s >= .@c)
		close;
	close2;
	warp .@m$[.@s], .@x[.@s], .@y[.@s];
	end;

OnInit:
	// Stand down when NPC Plus's Warper is on (see ChronWaykeeperCore).
	sleep 1000;
	if (getarraysize($@npcp_tmap$))
		disablenpc strnpcinfo(0);
	end;
}
